Data structure:
    node: val, next, prev
    stack s: s.head,s.tail,s.size

start(s)
    s.head <- NIL
    s.tail <- NIL
    s.size <- 0

isempty(s)
    if s.size = 0
        return 1
    else return 0

push(data)
    x <- allocate(node)
    if x = NIL
        print "Overflow"
        return
    x.val <- data
    x.next <- NIL
    x.prev <- s.tail
    if s.tail = NIL
        s.head <- x
    else
        s.tail.next <- x
    s.tail <- x
    s.size <- s.size + 1

pop()
    if isempty(s)
        print "underflow"
        return ERROR                    
    x <- s.tail
    v <- x.val
    s.tail <- x.prev
    if s.tail != NIL
        s.tail.next <- NIL
    else
        s.head <- NIL
    FREE(x)
    s.size <- s.size - 1
    return v

get_element_from_bottom(idx)
    if idx < 0 or idx >= s.size
        print "index out of range"
        return ERROR
    x <- s.head
    for i <- 1 to idx
        x <- x.next
    return x.val

get_element_from_top(idx)
    if idx<0 or idx>=s.size
        print "out of range"
        return ERROR
    x <- s.tail
    for i <- 1 to idx
        x <- x.prev
    return x.val

multiply()
    if s.size < 2
        print "underflow: not enough operands"
        return ERROR
    b <- pop()
    a <- pop()
    r <- a * b
    push(r)
    return r

divide()
    if s.size < 2
        print "underflow: not enough operands"
        return ERROR
    if s.tail.val = 0
        print "Error: division by zero"
        return ERROR
    b <- pop()
    a <- pop()
    q <- a / b                         
    if (a mod b != 0) and ((a < 0) xor (b < 0))
        q <- q - 1                    
    push(q)
    return q

add()
    if s.size < 2
        print "underflow: not enough operands"
        return ERROR
    b <- pop()                         
    a <- pop()                          
    r <- a + b
    push(r)
    return r

subtract()
    if s.size < 2
        print "underflow: not enough operands"
        return ERROR
    b <- pop()
    a <- pop()
    r <- a - b                         
    push(r)
    return r

calculate(eq, result)
    start(s)
    n <- length[eq]
    i <- 0
    while i < n
        c <- eq[i]
        if c = ' '
            i <- i + 1
            continue
        if (c >= '0' and c <= '9') or (c = '-' and i + 1 < n and eq[i+1] >= '0' and eq[i+1] <= '9')
            negative <- false
            if c = '-'
                negative <- true
                i <- i + 1
            val <- 0
            while i < n and eq[i] >= '0' and eq[i] <= '9'
                val <- val * 10 + (eq[i] - '0')
                i <- i + 1
            if negative
                val <- -val
            push(val)
        elseif c = '+' or c = '-' or c = '*' or c = '/'
            if s.size < 2
                print "underflow: not enough operands"
                destroy(s)
                return false
            if c = '/' and s.tail.val = 0
                print "Error: division by zero"
                destroy(s)
                return false
            if c = '+'
                add()
            elseif c = '*'
                multiply()
            elseif c = '-'
                subtract()
            else
                divide()
            i <- i + 1
        else
            print "invalid characters"
            destroy(s)
            return false
    if s.size != 1
        print "Error: Wrong equation"
        destroy(s)
        return false
    result <- pop()
    destroy(s)
    return true

destroy(s)
    x <- s.head
    while x != NIL
        temp <- x.next
        FREE(x)
        x <- temp
    s.head <- NIL
    s.tail <- NIL
    s.size <- 0
