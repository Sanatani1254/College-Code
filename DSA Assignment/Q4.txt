Data structure:
    stack s: s.val (pointer to an array), s.top, s.max
    min = 1024

start(s)
    s.val <- allocate(1024)
    s.top <- -1  
    s.max <- 1024

destroy(s)
    FREE(s.val)

isempty(s)
    return s.top = -1

resize(s, newCap)
    A <- allocate(newCap)
    if A = NIL
        return false
    for i <- 0 to s.top
        A[i] <- s.val[i]
    FREE(s.val)
    s.val <- A
    s.max <- newCap
    return true

pop(s,v)                                   
    if isempty(s)
        return false
    v <- s.val[s.top]
    s.top <- s.top - 1
    if s.max > min and s.top + 1 <= s.max / 4
        newCap <- max(min, s.max / 2)
        resize(s, newCap)                   
    return true

push(s,v)
    if s.top + 1 = s.max                      
        if resize(s,s.max*2) = false
            return false                    
    s.top <- s.top+1
    s.val[s.top] <- v
    return true

calculate (eq, result)                        
    start(s)
    n <- length[eq]
    i <- 0
    while i < n
        c <- eq[i]
        if c = ' '
            i <- i + 1
            continue
        isNum <- (c >= '0' and c <= '9') or (c = '-' and i + 1 < n and eq[i+1] >= '0' and eq[i+1] <= '9')
        if isNum
            negative <- false
            if c = '-'
                negative <- true
                i <- i + 1
            val <- 0
            while i < n and eq[i] >= '0' and eq[i] <= '9'
                val <- val * 10 + (eq[i] - '0')
                i <- i + 1
            if negative
                val <- -val
            if push(s, val) = false
                print "overflow"
                destroy(s)
                return false
        elseif c = '+' or c = '-' or c = '*' or c = '/' 
            if pop(s,b) = false or pop(s,a) = false
                print "underflow:Not enough oparands"
                destroy(s)
                return false
            if c = '+'
                r <- a + b
            elseif c = '*'
                r <- a * b
            elseif c = '-'
                r <- a - b
            else
                if b = 0
                    print "Error: division by zero"
                    destroy(s)
                    return false
                if c = '/'
                    r <- a / b              
            push(s, r)
            i <- i + 1
        else
            print "invalid characters"
            destroy(s)
            return false
    if pop(s, result) = false
        print "empty equation"
        destroy(s)
        return false
    if isempty(s) = false
        print "Error: Wrong equation"
        destroy(s)
        return false
    destroy(s)
    return true
